        ma_bai = 0;

    }

        ma_bai = ma;

        co1 = false;
        co2 = false;
        co3 = false;

        int dem = 0;
if (co1) dem++;

        if (co3) dem++;

        if (co1 && pt1.lay_ma_xe() == ma_xe)
            return true;

        if (co2 && pt2.lay_ma_xe() == ma_xe)
            return true;

        if (co3 && pt3.lay_ma_xe() == ma_xe)
            return true;

        return false;
    }

            co2 = true;
            return true;
        }

        if (kich_thuoc == 'L' && !co3) {
            pt3 = v;
            co3 = true;
            return true;
        }

        return false;
